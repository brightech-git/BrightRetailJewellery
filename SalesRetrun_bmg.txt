)

join cte on cte.TRFNO = RECEIPT.TRFNO and cte.CATCODE = RECEIPT.CATCODE

)
,cte3 as(
select cte2.*,LOTISSUE.PCS LIPCS,LOTISSUE.GRSWT LIGRSWT,LOTISSUE.LOTSNO from cte2

)
,cte4 as(

)

inner join BMGADMINDB..DESIGNER on ITEMTAG.DESIGNERID = DESIGNER.DESIGNERID
